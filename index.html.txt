<!-- SCRIPT UNTUK MENGIRIM DATA KE GOOGLE APPS SCRIPT -->
<script>
  // MAS BRO, PASTE URL WEB APP DARI GOOGLE APPS SCRIPT DI BAWAH INI:
  const API_URL = "https://script.google.com/macros/s/AKfycbwrptLezBuHJmmXDVVj9MSfLckwH89bM1IpZq4qDsrprP9gzvIc8inYI9jQtPkmvGJ4/exec";

  document.getElementById('formRegistrasi').addEventListener('submit', function(e) {
    e.preventDefault(); // Mencegah halaman reload
    
    const btn = document.getElementById('btn-submit');
    btn.innerHTML = '<span class="spinner-border spinner-border-sm"></span> Memproses...';
    btn.disabled = true;

    // Menyiapkan data untuk dikirim ke API
    const formData = new URLSearchParams();
    formData.append('action', 'register');
    formData.append('schoolId', document.getElementById('schoolId').value);
    formData.append('namaLengkap', document.getElementById('namaLengkap').value);
    formData.append('jabatan', document.getElementById('jabatan').value);
    formData.append('email', document.getElementById('email').value);

    // Proses pengiriman ke Google Apps Script
    fetch(API_URL, {
      method: 'POST',
      body: formData
    })
    .then(response => response.json())
    .then(result => {
      if(result.success) {
        // Jika sukses, hilangkan form, munculkan kartu QR
        document.getElementById('form-container').classList.add('d-none');
        document.getElementById('result-container').classList.remove('d-none');
        
        // Isi data ke dalam kartu
        document.getElementById('res-nama').innerText = result.nama;
        document.getElementById('res-jabatan').innerText = result.jabatan;
        document.getElementById('res-id').innerText = result.participantId;
        document.getElementById('res-qr').src = result.qrUrl;
      } else {
        alert("Gagal: " + result.message);
        btn.innerHTML = 'Daftar Sekarang';
        btn.disabled = false;
      }
    })
    .catch(error => {
      alert("Terjadi kesalahan jaringan.");
      btn.innerHTML = 'Daftar Sekarang';
      btn.disabled = false;
    });
  });
</script>
