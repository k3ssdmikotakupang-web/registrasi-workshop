function renderTabel() { let selectedSub = document.getElementById('filterSubrayon').value; let filtered = dataGlobal; // Logika Saringan Cerdas di Browser if(userData.role === 'Bendahara_Subrayon') { let subAkurat = userData.subrayon.toLowerCase().replace(/\s+/g, ''); filtered = dataGlobal.filter(d => { let subSekDB = (d.subrayon || '').toLowerCase().replace(/\s+/g, ''); return subSekDB === subAkurat || subSekDB.includes(subAkurat) || subAkurat.includes(subSekDB); }); } else if (selectedSub !== 'Semua') { filtered = dataGlobal.filter(d => d.subrayon === selectedSub); } let html = ''; let sNegeriL=0, sNegeriB=0, sSwastaL=0, sSwastaB=0; let sSDL=0, sSDB=0, sMIL=0, sMIB=0; if(filtered.length === 0) { html = 'Belum ada data pendaftaran di wilayah ini. Pastikan kolom Subrayon di DB_Sekolah sudah terisi.'; } else { filtered.forEach(d => { let badge = d.status_bayar === 'Lunas' ? 'Lunas' : 'Belum Bayar'; let isLunas = d.status_bayar === 'Lunas'; let st = (d.status_sekolah || '').toLowerCase(); let jen = (d.jenjang || '').toLowerCase(); if(st.includes('negeri')) { isLunas ? sNegeriL++ : sNegeriB++; } else { isLunas ? sSwastaL++ : sSwastaB++; } if(jen.includes('sd')) { isLunas ? sSDL++ : sSDB++; } else { isLunas ? sMIL++ : sMIB++; } html += ` ${d.npsn} ${d.nama} ${d.subrayon} ${d.jenjang} - ${d.status_sekolah} ${badge} `; }); } document.getElementById('tabelRekap').innerHTML = html; document.getElementById('stat-n-lunas').innerText = sNegeriL; document.getElementById('stat-n-belum').innerText = sNegeriB; document.getElementById('stat-s-lunas').innerText = sSwastaL; document.getElementById('stat-s-belum').innerText = sSwastaB; document.getElementById('stat-sd-lunas').innerText = sSDL; document.getElementById('stat-sd-belum').innerText = sSDB; document.getElementById('stat-mi-lunas').innerText = sMIL; document.getElementById('stat-mi-belum').innerText = sMIB; }
